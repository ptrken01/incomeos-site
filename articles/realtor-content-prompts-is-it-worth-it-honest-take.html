```python

import openai

def generate_listing_prompt(property_data):

prompt = f"""

Write a compelling real estate listing for:

Property Type: {property_data['type']}

Address: {property_data['address']}

Bedrooms: {property_data['bedrooms']}

Bathrooms: {property_data['bathrooms']}

Square Feet: {property_data['sqft']}

Price: ${property_data['price']:,}

Year Built: {property_data['year']}

- A captivating opening paragraph

- Key features (updated kitchen, backyard, etc.)

- Market positioning

- Call-to-action for inquiries

Keep it professional and engaging.

"""

return prompt
```
