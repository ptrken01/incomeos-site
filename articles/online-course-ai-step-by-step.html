```python

Sample Python script to automate prompt-based content generation

import openai

import json

def generate_course_content(prompt_template, course_data):

response = openai.ChatCompletion.create(

model="gpt-4",

messages=[

{"role": "system", "content": "You are an expert online course creator."},

{"role": "user", "content": prompt_template.format(**course_data)}

]

)

return response.choices[0].message.content

Example usage

course_outline_prompt = """

Create a detailed course outline for {topic} with {num_lessons} lessons.

Each lesson should have a title, duration (in minutes), and key learning objectives.

Format as JSON with keys: title, duration, objectives, content.

"""

course_data = {

"topic": "Python Data Science",

"num_lessons": 12

}

outline = generate_course_content(course_outline_prompt, course_data)

print(json.loads(outline))

```
