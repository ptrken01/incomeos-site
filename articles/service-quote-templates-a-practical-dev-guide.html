Service Quote Templates: A Practical Dev Guide

When building local service businesses, the ability to quickly generate professional quotes is crucial. This guide shows you how to create a reusable system for generating service quotes using simple templates and AI prompts.

The Template System

Here's a working Python script that generates quote templates from structured data:

```python

import json

from datetime import datetime

def generate_quote_template(service_data):

template = """

{company_name} - Service Quote

Date: {quote_date}

Quote #: {quote_number}

Client: {client_name}

Address: {client_address}

Services:

{services_list}

Total Amount: ${total_amount}

Payment Terms: {payment_terms}

Valid Until: {valid_until}

Thank you for your business!

{company_signature}

"""

services = "\n".join([f"- {s['name']}: ${s['price']}" for s in service_data['services']])

return template.format(

company_name=service_data.get('company_name', 'Your Company'),

quote_date=datetime.now().strftime("%Y-%m-%d"),

quote_number=service_data.get('quote_number', 'QT-001'),

client_name=service_data.get('client_name', ''),

client_address=service_data.get('client_address', ''),

services_list=services,

total_amount=service_data.get('total_amount', 0),

payment_terms=service_data.get('payment_terms', 'Net 30'),

valid_until=datetime.now().strftime("%Y-%m-%d"),

company_signature=service_data.get('signature', '')

)

Usage example

data = {

"company_name": "ABC Plumbing",

"quote_number": "QT-2024-001",

"client_name": "John Smith",

"client_address": "123 Main St, Anytown",

"services": [

{"name": "Leak Repair", "price": 150},

{"name": "Pipe Installation", "price": 300}

],

"total_amount": 450,

"payment_terms": "Net 15",

"signature": "Sarah Johnson, Owner"

}

print(generate_quote_template(data))

```

This system produces consistent, professional quotes that can be customized with any service data. The script handles date formatting, service listing, and total calculation automatically.

Advanced Template Integration

For businesses managing multiple service types, extend the template system to include service categories:

```python

def generate_categorized_quote(service_data):

# Group services by category

categories = {}

for service in service_data['services']:

cat = service.get('category', 'General')

if cat not in categories:

categories[cat] = []

categories[cat].append(service)

# Build categorized services list

services_list = ""

for category, services in categories.items():

services_list += f"\n{category}:\n"

for service in services:

services_list += f" - {service['name']}: ${service['price']}\n"

return generate_quote_template({**service_data, 'services_list': services_list})

```
