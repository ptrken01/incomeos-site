import openai

from datetime import datetime

def generate_client_program(client_data, prompt_template):

"""

Generate personalized training program using AI prompts

Input: client_data dict with goals, fitness level, preferences

Output: structured workout plan

"""

prompt = f"""

{prompt_template}

Client Profile:

- Name: {client_data['name']}

- Goal: {client_data['goal']}

- Fitness Level: {client_data['level']}

- Preferences: {client_data['preferences']}

Generate a 4-week program with weekly breakdowns.

"""

response = openai.ChatCompletion.create(

model="gpt-4",

messages=[{"role": "user", "content": prompt}],

temperature=0.7

)

return response.choices[0].message.content
